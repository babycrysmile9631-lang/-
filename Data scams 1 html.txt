    let activeTile = null;
    function setBaseLayer(key, btnEl) {
      document.querySelectorAll(".layer-btn").forEach(b => b.classList.remove("active"));
      if (btnEl) btnEl.classList.add("active");

      if (activeTile && map.hasLayer(activeTile)) {
        map.removeLayer(activeTile);
      }
      if (key === "grid") {
        activeTile = null;
        return;
      }
      activeTile = tileLayers[key] || tileLayers.osm;
      activeTile.addTo(map);
    }

    // Load default layer with safe fallback
    activeTile = tileLayers.osm.addTo(map);

    const overlayGroup = L.layerGroup().addTo(map);
    let markersById = {};

    function formatCoordsDisplay(lat, lon) {
      // Approximate UTM 48N & MGRS helper for Preah Vihear / Cambodia
      const easting = Math.round(500000 + (lon - 105.0) * 108200);
      const northing = Math.round(lat * 110574);
      const mgrsE = String(Math.max(0, easting % 100000)).padStart(5, "0");
      const mgrsN = String(Math.max(0, northing % 100000)).padStart(5, "0");
      document.getElementById("liveLatLon").textContent = `${lat.toFixed(5)}, ${lon.toFixed(5)}`;
      document.getElementById("liveUtm").textContent = `48N ${easting}E ${northing}N`;
      document.getElementById("liveMgrs").textContent = `48PUU${mgrsE}${mgrsN}`;
    }

    function renderAll() {
      overlayGroup.clearLayers();
      markersById = {};
      const listEl = document.getElementById("callList");
      listEl.innerHTML = "";
      document.getElementById("callCount").textContent = callRecords.length;

      const latLngs = [];

      callRecords.forEach((rec, idx) => {
        const isLast = idx === callRecords.length - 1;
        const pt = [rec.lat, rec.lon];
        latLngs.push(pt);

        // Coverage circle
        L.circle(pt, {
          radius: isLast ? 1500 : 1100,
          color: isLast ? "#ef4444" : "#f59e0b",
          weight: 1.5,
          fillColor: isLast ? "#ef4444" : "#f59e0b",
          fillOpacity: isLast ? 0.14 : 0.06
        }).addTo(overlayGroup);

        // Marker
        let marker;
        if (isLast) {
          const icon = L.divIcon({
            className: "",
            html: `<div class="target-marker-wrap"><div class="target-pulse"></div><div class="target-core"></div></div>`,
            iconSize: [24, 24],
            iconAnchor: [12, 12]
          });
          marker = L.marker(pt, { icon });
        } else {
          marker = L.circleMarker(pt, {
            radius: 7,
            color: "#ffffff",
            weight: 1.5,
            fillColor: "#0ea5e9",
            fillOpacity: 0.95
          });
        }

        marker.bindPopup(`
          <div style="color:#0f172a;font-size:12px;line-height:1.5;">
            <b>Point ${rec.id} (${rec.time})</b> ${isLast ? '<span style="color:#dc2626;font-weight:700;">[គោលដៅចុងក្រោយ]</span>' : ''}<br/>
            ទៅលេខ: <b>${rec.to}</b> (${rec.dur}s)<br/>
            ឃុំ: <b>${rec.commune}</b> | Cell: <b>${rec.cell}</b><br/>
            Lat/Lon: <code>${rec.lat}, ${rec.lon}</code>
          </div>
        `);

        marker.on("click", () => {
          highlightCard(rec.id);
          formatCoordsDisplay(rec.lat, rec.lon);
        });

        marker.addTo(overlayGroup);
        markersById[rec.id] = marker;

        // Sidebar Card
        const card = document.createElement("div");
        card.className = "call-card" + (isLast ? " active" : "");
        card.id = "card-" + rec.id;
        card.innerHTML = `
          <div class="call-top">
            <span>${rec.time} → ${rec.to}</span>
            <span class="pt-pill ${isLast ? "target" : ""}">Pt.${rec.id}${isLast ? " ★" : ""}</span>
          </div>
          <div class="call-meta">
            ឃុំ: <b>${rec.commune}</b> • រយៈពេល: ${rec.dur}s<br/>
            Cell: ${rec.cell}
          </div>
          <div class="call-coords">${rec.lat}, ${rec.lon}</div>
        `;
        card.onclick = () => {
          highlightCard(rec.id);
          map.setView(pt, 14);
          marker.openPopup();
          formatCoordsDisplay(rec.lat, rec.lon);
        };
        listEl.appendChild(card);
      });

      if (latLngs.length > 1) {
        L.polyline(latLngs, {
          color: "#38bdf8",
          weight: 2.5,
          dashArray: "6, 6",
          opacity: 0.9
        }).addTo(overlayGroup);
      }

      const lastRec = callRecords[callRecords.length - 1];
      if (lastRec) formatCoordsDisplay(lastRec.lat, lastRec.lon);
    }

    function highlightCard(id) {
      document.querySelectorAll(".call-card").forEach(c => c.classList.remove("active"));
      const el = document.getElementById("card-" + id);
      if (el) el.classList.add("active");
    }

    function fitAllPoints() {
      const coords = callRecords.map(r => [r.lat, r.lon]);
      if (coords.length > 0) {
        map.fitBounds(L.latLngBounds(coords).pad(0.25));
      }
    }

    function switchTab(tab, btn) {
      document.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active"));
      if (btn) btn.classList.add("active");
      if (tab === "target") {
        const last = callRecords[callRecords.length - 1];
        if (last) {
          highlightCard(last.id);
          map.setView([last.lat, last.lon], 14);
          markersById[last.id]?.openPopup();
        }
      } else {
        fitAllPoints();
      }
    }

    map.on("mousemove", (e) => {
      formatCoordsDisplay(e.latlng.lat, e.latlng.lng);
    });

    function openModal() { document.getElementById("callModal").classList.add("open"); }
    function closeModal() { document.getElementById("callModal").classList.remove("open"); }

    function addCallRecord() {
      const time = document.getElementById("inTime").value.trim() || "18:00:00";
      const to = document.getElementById("inTo").value.trim() || "0719964515";
      const commune = document.getElementById("inCommune").value.trim() || "ROMTOM";

      const lon = parseFloat(document.getElementById("inLon").value);
      if (isNaN(lat) || isNaN(lon)) return alert("សូមបញ្ចូល Lat និង Lon ជាលេខត្រឹមត្រូវ។");

      callRecords.push({
        id: callRecords.length + 1,
        time, to, dur: 30, commune, cell, lat, lon
      });
      renderAll();
      fitAllPoints();
      closeModal();
    }

    function exportCSV() {
      const headers = ["Point", "Start_Time", "To_Phone", "Duration_Sec", "Commune", "Cell_ID", "Latitude", "Longitude"];
      const rows = callRecords.map(r => [r.id, r.time, r.to, r.dur, r.commune, r.cell, r.lat, r.lon]);
      const csv = [headers, ...rows].map(row => row.map(v => `"${String(v).replace(/"/g, '""')}"`).join(",")).join("\n");
      const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = "0975424488_call_trace_16Nov26.csv";
      a.click();
      URL.revokeObjectURL(url);
    }
